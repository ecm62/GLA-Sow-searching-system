function renderHistory(data) { const tbody = document.getElementById('tableBody'); tbody.innerHTML = ''; data.forEach(row => { const tr = document.createElement('tr'); // v29.0 嚴格對位實體數據輸出欄位 const displayCells = [ row[0], // Sow ID row[1], // Boar ID `W${row[4]}`, // Mate Week row[5], // Mating Date row[6], // Wean Wt. (kg) row[7], // Farrow Date `P${row[8]}`, // Parity row[9], // Still Born row[10], // Total Born row[11] // Born Live ]; displayCells.forEach((cell, index) => { const td = document.createElement('td'); td.textContent = (cell === undefined || cell === null) ? "-" : cell; // 數據可視化增強 if(index === 0) td.style.fontWeight = "800"; if(index === 6) td.className = "hl-parity"; if(index === 8) td.style.background = "#F0F9FF"; td.style.fontWeight = "800"; if(index === 9) { td.style.background = "#F0FDF4"; td.style.color = "#16A34A"; td.style.fontWeight = "800"; } tr.appendChild(td); }); tbody.appendChild(tr); }); }
